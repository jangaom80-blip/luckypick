// AC(Arithmetic Complexity) 값 계산 함수 function calculateAC(numbers) { const diffs = new Set(); for (let i = 0; i < numbers.length; i++) { for (let j = i + 1; j < numbers.length; j++) { diffs.add(Math.abs(numbers[i] - numbers[j])); } } return diffs.size - (numbers.length - 1); } // 필터 검증 파이프라인 function isValidCombination(numbers, options = {}) { const { oddRange = [2, 4], // 홀수 개수 2~4개 sumRange = [100, 175], // 총합 100~175 minAC = 7, // 산술적 복잡도 7 이상 maxConsecutive = 1 // 연속수 최대 1쌍 (3연번 금지) } = options; // 1. 홀짝 검증 const oddCount = numbers.filter(n => n % 2 !== 0).length; if (oddCount < oddRange[0] || oddCount > oddRange[1]) return false; // 2. 총합 검증 const sum = numbers.reduce((acc, cur) => acc + cur, 0); if (sum < sumRange[0] || sum > sumRange[1]) return false; // 3. 연속 번호 검증 (정렬 상태 필수) let consecutivePairs = 0; for (let i = 0; i < numbers.length - 1; i++) { if (numbers[i + 1] - numbers[i] === 1) { consecutivePairs++; if (i < numbers.length - 2 && numbers[i + 2] - numbers[i + 1] === 1) { return false; // 3연속 번호 발견 시 즉시 탈락 } } } if (consecutivePairs > maxConsecutive) return false; // 4. AC값 검증 if (calculateAC(numbers) < minAC) return false; // 5. 끝자리 중복 검증 (동일 끝자리 3개 이상 탈락) const lastDigits = numbers.map(n => n % 10); const digitCounts = {}; for (const d of lastDigits) { digitCounts[d] = (digitCounts[d] || 0) + 1; if (digitCounts[d] >= 3) return false; } return true; } // 최종 6개 번호 생성기 function generateLottoNumbers(includeNums = [], excludeNums = [], options = {}) { const MAX_ATTEMPTS = 200; let attempts = 0; const pool = Array.from({ length: 45 }, (_, i) => i + 1) .filter(n => !excludeNums.includes(n)); while (attempts < MAX_ATTEMPTS) { attempts++; const selected = [...includeNums]; const availablePool = pool.filter(n => !includeNums.includes(n)); // 남은 자리만큼 셔플 후 추출 while (selected.length < 6 && availablePool.length > 0) { const randIdx = Math.floor(Math.random() * availablePool.length); selected.push(availablePool.splice(randIdx, 1)[0]); } selected.sort((a, b) => a - b); // 필터 검증 통과 시 반환 if (isValidCombination(selected, options)) { return { numbers: selected, attempts }; } } // 조건이 너무 빡빡해 루프 초과 시 기본 셔플 조합 반환 (Fallback) return { numbers: pool.sort(() => 0.5 - Math.random()).slice(0, 6).sort((a, b) => a - b), fallback: true }; }
